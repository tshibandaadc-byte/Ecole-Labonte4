// =========================================
// lib/services/api_service.dart — Client HTTP centralisé
// =========================================
import 'package:dio/dio.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import '../utils/config.dart';

class ApiException implements Exception {
  final String message;
  final int? statusCode;
  ApiException(this.message, {this.statusCode});

  @override
  String toString() => message;
}

class ApiService {
  static final ApiService _instance = ApiService._interne();
  factory ApiService() => _instance;

  late final Dio dio;
  final _storage = const FlutterSecureStorage();
  static const _cleToken = 'jwt_token';

  ApiService._interne() {
    dio = Dio(BaseOptions(
      baseUrl: Config.baseUrl,
      connectTimeout: const Duration(seconds: 12),
      receiveTimeout: const Duration(seconds: 12),
      headers: {'Content-Type': 'application/json'},
    ));

    dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        final token = await _storage.read(key: _cleToken);
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        handler.next(options);
      },
      onError: (DioException erreur, handler) {
        handler.next(erreur);
      },
    ));
  }

  Future<void> sauverToken(String token) async {
    await _storage.write(key: _cleToken, value: token);
  }

  Future<String?> lireToken() async {
    return _storage.read(key: _cleToken);
  }

  Future<void> supprimerToken() async {
    await _storage.delete(key: _cleToken);
  }

  Future<bool> estConnecte() async {
    final token = await lireToken();
    return token != null;
  }

  /// Traduit les erreurs Dio en messages clairs pour l'utilisateur.
  ApiException traiterErreur(DioException e) {
    if (e.type == DioExceptionType.connectionTimeout ||
        e.type == DioExceptionType.receiveTimeout) {
      return ApiException("Le serveur ne répond pas. Vérifiez votre connexion.");
    }
    if (e.type == DioExceptionType.connectionError) {
      return ApiException("Impossible de joindre le serveur. Vérifiez votre connexion internet.");
    }
    final data = e.response?.data;
    final statusCode = e.response?.statusCode;
    String message = "Une erreur est survenue.";
    if (data is Map && data['detail'] != null) {
      message = data['detail'] is String ? data['detail'] : data['detail'].toString();
    } else if (data is Map && data['message'] != null) {
      message = data['message'].toString();
    }
    return ApiException(message, statusCode: statusCode);
  }
}

import 'package:dio/dio.dart';
import '../models/utilisateur.dart';
import '../utils/config.dart';
import 'api_service.dart';

class AuthService {
  final ApiService _api = ApiService();

  Future<Utilisateur> connecter(String identifiant, String motDePasse) async {
    try {
      final reponse = await _api.dio.post(
        "${Config.apiAuth}/login",
        data: {"identifiant": identifiant, "mot_de_passe": motDePasse},
      );

      final token = reponse.data['access_token'] as String;
      await _api.sauverToken(token);

      return Utilisateur.fromJson(reponse.data['utilisateur'] as Map<String, dynamic>);
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }

  Future<void> deconnecter() async {
    await _api.supprimerToken();
  }

  Future<bool> estConnecte() => _api.estConnecte();
}

import 'package:dio/dio.dart';
import '../models/eleve.dart';
import '../utils/config.dart';
import 'api_service.dart';

class EleveService {
  final ApiService _api = ApiService();

  Future<Map<String, dynamic>> inscrire({
    required String nom,
    required String prenom,
    required String dateNaissance, // format YYYY-MM-DD
    String? lieuNaissance,
    required String sexe,
    required int classeId,
    required String motDePasse,
    String? telephone,
    String? email,
    String? tuteurNom,
    String? tuteurTelephone,
  }) async {
    try {
      final reponse = await _api.dio.post(
        "${Config.apiEleves}/inscription",
        data: {
          "nom": nom,
          "prenom": prenom,
          "date_naissance": dateNaissance,
          "lieu_naissance": lieuNaissance,
          "sexe": sexe,
          "classe_id": classeId,
          "mot_de_passe": motDePasse,
          "telephone": telephone,
          "email": email,
          "tuteur_nom": tuteurNom,
          "tuteur_telephone": tuteurTelephone,
        },
      );
      return reponse.data['data'] as Map<String, dynamic>;
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }

  Future<Eleve> obtenirParId(int eleveId) async {
    try {
      final reponse = await _api.dio.get("${Config.apiEleves}/$eleveId");
      return Eleve.fromJson(reponse.data['data'] as Map<String, dynamic>);
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }

  Future<List<Eleve>> listerParClasse(int classeId) async {
    try {
      final reponse = await _api.dio.get("${Config.apiEleves}/classe/$classeId");
      final liste = reponse.data['data'] as List;
      return liste.map((j) => Eleve.fromJson(j as Map<String, dynamic>)).toList();
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }
}

class NoteService {
  final ApiService _api = ApiService();

  Future<Map<String, dynamic>> listerNotesEleve(int eleveId, {String? periode}) async {
    try {
      final reponse = await _api.dio.get(
        "${Config.apiNotes}/eleve/$eleveId",
        queryParameters: periode != null ? {"periode": periode} : null,
      );
      final data = reponse.data['data'] as Map<String, dynamic>;
      final notes = (data['notes'] as List)
          .map((j) => Note.fromJson(j as Map<String, dynamic>))
          .toList();
      return {"notes": notes, "moyenne": data['moyenne']};
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }

  Future<Bulletin> obtenirBulletin(int eleveId, String periode) async {
    try {
      final reponse = await _api.dio.get("${Config.apiBulletins}/$eleveId/$periode");
      return Bulletin.fromJson(reponse.data['data'] as Map<String, dynamic>);
    } on DioException catch (e) {
      throw _api.traiterErreur(e);
    }
  }

  Future<Note> saisirNote({
    required int eleveId,
    required int matiereId,
    required double note,
    required String periode,
    String? typeEval,
    int coefficient = 1,
  }) async {

import 'package:flutter/foundation.dart';
import '../models/utilisateur.dart';
import '../services/auth_service.dart';
import '../services/api_service.dart';

enum EtatAuth { inconnu, connecte, deconnecte }

class AuthProvider extends ChangeNotifier {
  final AuthService _authService = AuthService();

  Utilisateur? _utilisateur;
  EtatAuth _etat = EtatAuth.inconnu;
  bool _chargement = false;
  String? _erreur;

  Utilisateur? get utilisateur => _utilisateur;
  EtatAuth get etat => _etat;
  bool get chargement => _chargement;
  String? get erreur => _erreur;
  bool get estConnecte => _etat == EtatAuth.connecte;

  Future<void> verifierSessionExistante() async {
    final token = await ApiService().lireToken();
    if (token != null) {
      _etat = EtatAuth.connecte;
    } else {
      _etat = EtatAuth.deconnecte;
    }
    notifyListeners();
  }

  Future<bool> connecter(String identifiant, String motDePasse) async {
    _chargement = true;
    _erreur = null;
    notifyListeners();

    try {
      final utilisateur = await _authService.connecter(identifiant, motDePasse);
      _utilisateur = utilisateur;
      _etat = EtatAuth.connecte;
      _chargement = false;
      notifyListeners();
      return true;
    } catch (e) {
      _erreur = e.toString();
      _chargement = false;
      _etat = EtatAuth.deconnecte;
      notifyListeners();
      return false;
    }
  }

  Future<void> deconnecter() async {
    await _authService.deconnecter();
    _utilisateur = null;
    _etat = EtatAuth.deconnecte;
    notifyListeners();
  }

  void effacerErreur() {
    _erreur = null;
    notifyListeners();
  }
}

class ConversationScreen extends StatefulWidget {
  final int conversationId;
  final int destinataireId;
  final String nomDestinataire;

  const ConversationScreen({
    super.key,
    required this.conversationId,
    required this.destinataireId,
    required this.nomDestinataire,
  });

  @override
  State<ConversationScreen> createState() => _ConversationScreenState();
}

class _ConversationScreenState extends State<ConversationScreen> {
  final _messageCtrl = TextEditingController();
  final _scrollCtrl = ScrollController();

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) async {
      await context.read<CommunicationProvider>().chargerMessages(widget.conversationId);
      _defilerVersLeBas();
    });
  }

  @override
  void dispose() {
    _messageCtrl.dispose();
    _scrollCtrl.dispose();
    super.dispose();
  }

  void _defilerVersLeBas() {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (_scrollCtrl.hasClients) {
        _scrollCtrl.jumpTo(_scrollCtrl.position.maxScrollExtent);
      }
    });
  }

  Future<void> _envoyer() async {
    final texte = _messageCtrl.text.trim();
    if (texte.isEmpty) return;
    _messageCtrl.clear();

    final succes = await context.read<CommunicationProvider>().envoyerMessage(
          widget.destinataireId, texte, widget.conversationId,
        );

    if (succes) _defilerVersLeBas();
  }

  @override
  Widget build(BuildContext context) {
    final provider = context.watch<CommunicationProvider>();
    final utilisateur = context.read<AuthProvider>().utilisateur;

    return Scaffold(
      appBar: AppBar(title: Text(widget.nomDestinataire)),
      body: Column(
        children: [
          Expanded(
            child: provider.chargement
                ? const Center(child: CircularProgressIndicator())
                : ListView.builder(
                    controller: _scrollCtrl,
                    padding: const EdgeInsets.all(12),
                    itemCount: provider.messagesConversation.length,
                    itemBuilder: (context, index) {
                      final m = provider.messagesConversation[index];
                      final estMoi = m.expediteurId == utilisateur?.id;
                      return Align(
                        alignment: estMoi ? Alignment.centerRight : Alignment.centerLeft,
                        child: Container(
                          margin: const EdgeInsets.symmetric(vertical: 4),

PROCHAINES ÉTAPES POSSIBLES :
1. Installer Flutter sur un PC (https://docs.flutter.dev/get-started/install)
2. Créer le projet : flutter create ecole_la_bonte
3. Recréer l'arborescence ci-dessus et coller chaque section dans son fichier
4. flutter pub get
5. flutter run (avec un émulateur ou téléphone branché en USB)
6. Une fois prêt : flutter build apk --release
   → le fichier .apk installable se trouve alors dans
     build/app/outputs/flutter-apk/app-release.apk
